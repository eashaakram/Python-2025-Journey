C++ — Linked List 
struct Node {
  int data;
  Node* next;
};

// Insert at front - O(1)
Node* insertFront(Node* head, int val) {
  Node* n = new Node();
  n->data = val;
  n->next = head;
  return n;
}

// Insert at end - O(n)
Node* insertEnd(Node* head, int val) {
  Node* n = new Node();
  n->data = val; n->next = NULL;
  if (!head) return n;
  Node* cur = head;
  while (cur->next) cur = cur->next;
  cur->next = n;
  return head;
}

// Delete at front - O(1)
Node* deleteFront(Node* head) {
  if (!head) return NULL;
  Node* temp = head;
  head = head->next;
  delete temp;
  return head;
}

📖 Explanation

Linked List is a linear data structure where each element (node) contains data and a pointer to the next node.

1

Insert at Front: O(1) — new node points to old head

2

Insert at End: O(n) — traverse to last node

3

Insert at Position: O(n) — traverse to position

4

Delete at Front: O(1) — advance head pointer

5

Delete at Position: O(n) — traverse to position-1
