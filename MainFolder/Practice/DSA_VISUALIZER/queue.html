C++ — Simple Queue 
class Queue {
  int arr[100], front, rear;
public:
  Queue() { front = rear = -1; }

  void enqueue(int val) {
    if (rear == 99) {
      cout << "Overflow"; return;
    }
    if (front == -1) front = 0;
    arr[++rear] = val;
  }

  void dequeue() {
    if (front == -1 || front > rear) {
      cout << "Underflow"; return;
    }
    cout << arr[front++];
  }
};

📖 Explanation — Simple Queue

Queue is a FIFO (First-In, First-Out) structure.

1

Enqueue: add element at REAR → O(1)

2

Dequeue: remove element at FRONT → O(1)

3

Front: view first element without removing → O(1)

C++ — Circular Queue 
class CircularQueue {
  int* arr; int front, rear, size;
public:
  CircularQueue(int s) {
    size = s; arr = new int[s];
    front = rear = -1;
  }
  void enqueue(int val) {
    if ((rear+1) % size == front) {
      cout << "Overflow"; return;
    }
    if (front == -1) front = 0;
    rear = (rear+1) % size;
    arr[rear] = val;
  }
};

📖 Explanation — Circular Queue

Circular Queue connects the last position back to the first, reusing empty slots.

1

rear = (rear + 1) % size

2

Overflow: (rear + 1) % size == front

3

Underflow: front == -1
